module gpio_reg_tb();
		reg sys_clk,sys_rstn,gpio_we,gpio_eclk;
		reg [31:0] gpio_addr,gpio_data_i,aux_out,in_pad_i;
		wire [31:0] out_pad_o,oen_padoe_n;
		wire [31:0] gpio_data_o;
		wire gpio_inta_o;

gpio_reg DUT(sys_clk,sys_rstn,gpio_we,gpio_eclk,gpio_addr,gpio_data_i,aux_out,in_pad_i,out_pad_o,oen_padoe_n,gpio_data_o,gpio_inta_o);

initial
begin
	sys_clk=1'b0;
	forever #5 sys_clk=~sys_clk;
end

initial
begin
	gpio_eclk=1'b0;
	forever #10 gpio_eclk=~gpio_eclk;
end

task reset();
	begin
		repeat(2)
		begin
		@(negedge sys_clk)
		sys_rstn=1'b0;
		@(negedge sys_clk)
		sys_rstn=1'b1;
	end
	end
endtask


task write(input [31:0] addr,data);
	begin
		@(negedge sys_clk)
		gpio_addr=addr;
		gpio_data_i=data;
		gpio_we=1'b1;
		@(negedge sys_clk)
		gpio_we=1'b0;
	
	end
endtask

task read(input [31:0] addr);
	begin
		@(negedge sys_clk)
		gpio_addr=addr;
		gpio_we=1'b0;
	end
endtask


task cntrl(input [31:0] a,b, input c);
	begin
		gpio_addr=a;
		gpio_we=c;
		gpio_data_i=b;
	end
endtask

task aux1(input [31:0] a);
	begin
		aux_out=a;
	end
endtask
