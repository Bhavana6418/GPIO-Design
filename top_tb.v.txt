module top_tb();

	   reg pclk,prstn,psel,penable,pwrite,ext_clk_pad_i;
	   reg [31:0] paddr,pwdata,aux_in;
	   wire pready,irq;
	   wire [31:0] prdata;
	   wire [31:0] io_pad;

top DUT(pclk,prstn,psel,penable,pwrite,ext_clk_pad_i,paddr,pwdata,aux_in,pready,irq,prdata,io_pad);

//generation of pclk
initial 
begin
		pclk=1'b0;
	forever #5 pclk=~pclk;
end


//generation of external clk
initial
begin
	ext_clk_pad_i=1'b0;
	forever #7 ext_clk_pad_i=~ext_clk_pad_i;
end

//task for reset

task reset;
	begin	
	//repeat()
//begin
		@(negedge pclk)
		prstn=1'b0;
		@(negedge pclk)
		prstn=1'b1;
//	end
end
endtask

//initialize all input to zero
task initialize;
	begin
		@(negedge pclk)
		psel=0;
		pwrite=0;
		penable=0;
		paddr=0;
		pwdata=0;
		aux_in=0;
	end
endtask

//auxilary inputs for gpio core
task aux_write(input [31:0] in);
	begin
		@(negedge pclk)
		aux_in=in;
	end
endtask

//write into register as per apb protocol
task apb_write(input [31:0] addr,in);
begin
		@(negedge pclk)
		paddr=addr;
		psel=1'b1;
		pwrite=1'b1;
		penable=1'b0;
		pwdata=in;

		@(negedge pclk)
		paddr=addr;
		psel=1'b1;
		pwrite=1'b1;
		penable=1'b1;
		pwdata=in;

		@(negedge pclk)
		paddr=addr;
		psel=1'b1;
		pwrite=1'b1;
		penable=1'b0;
		pwdata=in;

		@(negedge pclk)
		paddr=addr;
		psel=1'b0;
		pwrite=1'b1;
		penable=1'b0;
		pwdata=in;
	end
endtask

//read from the registers as per the apb protocol
task apb_read(input [31:0] addr);
	begin
		@(negedge pclk)
		paddr=addr;
		psel=1'b1;
		pwrite=1'b0;
		penable=1'b0;
		
		@(negedge pclk)
		paddr=addr;
		psel=1'b1;
		pwrite=1'b0;
		penable=1'b1;
		
		@(negedge pclk)
		paddr=addr;
		psel=1'b1;
		pwrite=1'b0;
		penable=1'b0;
		
		@(negedge pclk)
		paddr=addr;
		psel=1'b0;
		pwrite=1'b0;
		penable=1'b0;
	end
endtask

//io interface as bidirectional

reg io_dir=1'b1; //used s input
reg [31:0] temp=32'bz;

//io_pad inputs
task io_pad_inputs(input [31:0]in_temp);
	begin
		io_dir=1'b0;
		temp=in_temp;
	end
endtask

//io_pad outputs
task io_pad_outputs;
	begin
		io_dir=1'b1;
	end
endtask

assign io_pad=~io_dir?temp:32'bz;
